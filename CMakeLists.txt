cmake_minimum_required(VERSION 3.30)
project(Euphony VERSION 0.2.0 LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "Euphony requires Windows TSF and the Windows SDK")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_definitions(EUPHONY_VERSION="${PROJECT_VERSION}")
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
option(EUPHONY_TSF_ONLY "Build only the 32-bit TSF shim for an existing 64-bit broker" OFF)
set(PHONO_CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../phono-core" CACHE PATH "Local phono-core checkout")
set(WEBVIEW2_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/webview2" CACHE PATH "Official WebView2 SDK")
if(DEFINED ENV{CONDA_PREFIX})
    list(PREPEND CMAKE_PREFIX_PATH "$ENV{CONDA_PREFIX}/Library")
endif()
find_package(nlohmann_json CONFIG REQUIRED)
include(CTest)

add_library(euphony_common STATIC src/common/win_util.cpp src/common/ipc.cpp src/common/config.cpp)
target_include_directories(euphony_common PUBLIC src)
target_link_libraries(euphony_common PUBLIC nlohmann_json::nlohmann_json user32 advapi32 shell32 ole32)
cmake_path(ABSOLUTE_PATH PHONO_CORE_DIR NORMALIZE OUTPUT_VARIABLE core_absolute)
cmake_path(GET core_absolute PARENT_PATH collection_absolute)
target_compile_definitions(euphony_common PRIVATE EUPHONY_COLLECTION_DIR="${collection_absolute}")

file(GLOB tsf_sources CONFIGURE_DEPENDS src/tsf/*.cpp)
add_library(EuphonyTsf SHARED ${tsf_sources} src/tsf/euphony.def src/euphony.rc)
target_include_directories(EuphonyTsf PRIVATE img)
target_link_libraries(EuphonyTsf PRIVATE euphony_common uuid gdi32 oleaut32)
set_target_properties(EuphonyTsf PROPERTIES PREFIX "")
if(BUILD_TESTING)
    add_executable(tsf_policy_tests tests/tsf_policy_tests.cpp)
    target_include_directories(tsf_policy_tests PRIVATE src)
    add_test(NAME tsf_policy COMMAND tsf_policy_tests)
    add_executable(tsf_smoke_tests tests/tsf_smoke_tests.cpp)
    target_include_directories(tsf_smoke_tests PRIVATE src)
    target_link_libraries(tsf_smoke_tests PRIVATE euphony_common uuid oleaut32)
    add_test(NAME tsf_smoke COMMAND tsf_smoke_tests $<TARGET_FILE:EuphonyTsf>)
    set_tests_properties(tsf_smoke PROPERTIES TIMEOUT 30 ENVIRONMENT "EUPHONY_USER_DIR=${CMAKE_BINARY_DIR}/tsf-test-profile;EUPHONY_INSTANCE=tsf-smoke-${CMAKE_SIZEOF_VOID_P}")
endif()

if(EUPHONY_TSF_ONLY)
    install(TARGETS EuphonyTsf RUNTIME DESTINATION x86 LIBRARY DESTINATION x86)
else()
    list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/.deps/vcpkg-installed/x64-windows-static")
    find_package(kenlm CONFIG REQUIRED)
    file(GLOB engine_sources CONFIGURE_DEPENDS src/engine/*.cpp)
    add_library(euphony_engine STATIC ${engine_sources})
    target_include_directories(euphony_engine PUBLIC src "${PHONO_CORE_DIR}/interface")
    target_link_libraries(euphony_engine PUBLIC euphony_common kenlm::kenlm)
    add_executable(EuphonyBroker WIN32 src/broker/main.cpp src/broker/tray.cpp src/euphony.rc)
    target_include_directories(EuphonyBroker PRIVATE img)
    target_link_libraries(EuphonyBroker PRIVATE euphony_engine imm32 uxtheme)
    # Keep Unicode KenLM paths local to our own process; injected TSF DLLs
    # must not change the host application's code page.
    set(utf8_manifest "${CMAKE_CURRENT_SOURCE_DIR}/src/broker/utf8.manifest")
    target_link_options(EuphonyBroker PRIVATE "/MANIFESTINPUT:${utf8_manifest}")
    set_property(TARGET EuphonyBroker APPEND PROPERTY LINK_DEPENDS "${utf8_manifest}")

    add_executable(EuphonySettings WIN32 src/settings/main.cpp src/euphony.rc)
    target_include_directories(EuphonySettings PRIVATE img)
    set_property(TARGET EuphonySettings APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/settings/settings.html")
    target_include_directories(EuphonySettings PRIVATE "${WEBVIEW2_SDK_DIR}/build/native/include")
    target_link_libraries(EuphonySettings PRIVATE euphony_common gdi32 uuid comctl32 dwmapi gdiplus)
    add_custom_command(TARGET EuphonySettings POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_CURRENT_SOURCE_DIR}/src/settings/settings.html" "$<TARGET_FILE_DIR:EuphonySettings>/settings.html"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${WEBVIEW2_SDK_DIR}/build/native/x64/WebView2Loader.dll"
        "$<TARGET_FILE_DIR:EuphonySettings>/WebView2Loader.dll")
    if(EXISTS "${PHONO_CORE_DIR}/results/phono_core.dll")
        add_custom_command(TARGET EuphonyBroker POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${PHONO_CORE_DIR}/results/phono_core.dll" "$<TARGET_FILE_DIR:EuphonyBroker>/phono_core.dll")
    endif()
    add_custom_command(TARGET EuphonyBroker POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory
        "$<TARGET_FILE_DIR:EuphonyBroker>/data" COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_CURRENT_SOURCE_DIR}/data/libime.tsv" "$<TARGET_FILE_DIR:EuphonyBroker>/data/libime.tsv")
    add_custom_command(TARGET EuphonyBroker POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_CURRENT_SOURCE_DIR}/data/zh_CN.lm" "$<TARGET_FILE_DIR:EuphonyBroker>/data/zh_CN.lm")
    install(TARGETS EuphonyTsf EuphonyBroker EuphonySettings RUNTIME DESTINATION . LIBRARY DESTINATION .)
    install(FILES src/settings/settings.html "${WEBVIEW2_SDK_DIR}/build/native/x64/WebView2Loader.dll" DESTINATION .)
    install(FILES "${PHONO_CORE_DIR}/results/phono_core.dll" DESTINATION .)
    install(FILES "${PHONO_CORE_DIR}/LICENSE" DESTINATION licenses RENAME PHONO-CORE-APACHE-2.0.txt)
    install(FILES data/libime.tsv data/zh_CN.lm DESTINATION data)
    install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/.deps/vcpkg-installed/x64-windows-static/share/" DESTINATION licenses/vcpkg
        FILES_MATCHING PATTERN copyright)
    install(FILES data/README.md DESTINATION data)
    install(FILES "${WEBVIEW2_SDK_DIR}/LICENSE.txt" DESTINATION licenses RENAME WebView2-LICENSE.txt)
    install(FILES "${WEBVIEW2_SDK_DIR}/NOTICE.txt" DESTINATION licenses RENAME WebView2-NOTICE.txt)
    install(DIRECTORY data/licenses/ DESTINATION licenses OPTIONAL)
    install(FILES README.md LICENSE THIRD_PARTY.md DESTINATION .)
    install(DIRECTORY img/ DESTINATION img)
    install(FILES tools/manage.ps1 tools/register.ps1 tools/unregister.ps1 tools/native_registration.ps1 DESTINATION tools)
    if(BUILD_TESTING)
        add_executable(common_tests tests/common_tests.cpp)
        target_link_libraries(common_tests PRIVATE euphony_common)
        add_test(NAME common COMMAND common_tests)
        if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/engine_tests.cpp")
            add_executable(engine_tests tests/engine_tests.cpp)
            target_link_libraries(engine_tests PRIVATE euphony_engine)
            target_link_options(engine_tests PRIVATE "/MANIFESTINPUT:${utf8_manifest}")
            set_property(TARGET engine_tests APPEND PROPERTY LINK_DEPENDS "${utf8_manifest}")
            target_compile_definitions(engine_tests PRIVATE EUPHONY_TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")
            add_test(NAME engine COMMAND engine_tests)
            set_tests_properties(engine PROPERTIES TIMEOUT 90)
        endif()
    endif()
endif()
